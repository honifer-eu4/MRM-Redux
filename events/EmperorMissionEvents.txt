# The Geneva Conventions
country_event = {
	id = emp_mission_events.3
	title = emp_mission_events.3.t
	desc = emp_mission_events.3.d
	picture = RELIGIOUS_CONVERSION_eventPicture
	
	is_triggered_only = yes
	major = yes
	
	immediate = {
		hidden_effect = {
			random_country = {
				limit = {
					OR = {
						religion = protestant
						religion = reformed
					}
					capital_scope = {
						continent = europe
					}
				}
				save_event_target_as = geneva_conventions_1
				set_country_flag = geneva_conventions_target
			}
			random_country = {
				limit = {
					OR = {
						religion = protestant
						religion = reformed
					}
					capital_scope = {
						continent = europe
					}
					NOT = { has_country_flag = geneva_conventions_target }
					has_regency = no
				}
				set_country_flag = geneva_conventions_target
				save_event_target_as = geneva_conventions_2
			}
			random_country = {
				limit = {
					OR = {
						religion = protestant
						religion = reformed
					}
					capital_scope = {
						continent = europe
					}
					NOT = { has_country_flag = geneva_conventions_target }
				}
				save_event_target_as = geneva_conventions_3
			}
		}
	}
	
	option = {
		name = emp_mission_events.3.a
		custom_tooltip = geneva_religious_center
		4720 = {
			add_province_triggered_modifier = religious_center
		}
		every_country = {
			limit = {
				OR = {
					religion = protestant
					religion = reformed
					religion = anglican
					religion = hussite
				}
				capital_scope = {
					continent = europe
				}
				NOT = { tag = SWI }
			}
			add_opinion = {
				who = ROOT
				modifier = opinion_geneva_conventions
			}
		}
	}
}
